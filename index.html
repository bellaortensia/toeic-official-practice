③ ノート保存用スプレッドシート(任意)
  1. Googleスプレッドシートを新規作成する(1枚のシートでOK。「Notes」タブは自動作成されます)。
  2. メニュー「拡張機能」→「Apps Script」を開く。
  3. デフォルトのコードを全て削除し、下のコードを貼り付けて保存。
  4. 右上「デプロイ」→「新しいデプロイ」→種類「ウェブアプリ」。実行ユーザー「自分」、アクセスできるユーザー「全員」に設定してデプロイ。
  5. 発行されたURL(.../execで終わるもの)をコピーし、上の「APPS SCRIPT WEB APP URL」欄に貼り付ける。
  6. これでヘッダーの「💾 保存」ボタンを押すと、今開いているノートがこのスプレッドシートに保存され、次回同じ問題/パッセージを開いたときに自動で復元されます(URL未設定の場合はこの端末のブラウザ内にのみ保存されます)。
function _getNotesSheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('Notes'); if (!sheet) { sheet = ss.insertSheet('Notes'); sheet.appendRow(['key', 'html', 'updatedAt']); } return sheet; } function _getAllNotes(sheet) { var last = sheet.getLastRow(); var result = {}; if (last >= 2) { var data = sheet.getRange(2, 1, last - 1, 2).getValues(); for (var i = 0; i < data.length; i++) { if (data[i][0]) result[data[i][0]] = data[i][1]; } } return result; } function _setNoteValue(sheet, key, html) { var last = sheet.getLastRow(); if (last >= 2) { var data = sheet.getRange(2, 1, last - 1, 1).getValues(); for (var i = 0; i < data.length; i++) { if (data[i][0] === key) { sheet.getRange(i + 2, 2, 1, 2).setValues([[html, new Date()]]); return; } } } sheet.appendRow([key, html, new Date()]); } function doGet(e) { var action = e.parameter.action; if (action === 'getNotes') { var sheet = _getNotesSheet(); return ContentService.createTextOutput(JSON.stringify({ notes: _getAllNotes(sheet) })) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ error: 'unknown action' })) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var body = JSON.parse(e.postData.contents); if (body.action === 'saveNotes') { var sheet = _getNotesSheet(); var notes = body.notes || {}; for (var key in notes) { _setNoteValue(sheet, key, notes[key]); } return ContentService.createTextOutput(JSON.stringify({ status: 'ok', count: Object.keys(notes).length })) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ error: 'unknown action' })) .setMimeType(ContentService.MimeType.JSON); }
